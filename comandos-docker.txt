# Check port:
netstat -oaf | findstr5432

# Create network
docker create network library-network

# Rodar container Postgre na versão 18.7
docker run --name libraryDB -p 5432:5432 -e POSTGRES_PASSWORD=postgres -e POSTGRES_USER=postgres -e POSTGRES_DB=library --network library-network postgres:18.7

# Rodar Pgadmin4
docker run --name pdadmin4 -p 15432:80 -e PGADMIN_DEFAULT_EMAIL=admin@admin.com -e PGADMIN_DEFAULT_PASSWORD=admin --network library-network dpage/pgadmin4
